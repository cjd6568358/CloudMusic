## 1. 职责与边界

| 做 | 不做 |
|---|---|
| 定义 `Obj / File / Link / Storage / Addition / Track` 模型 | 不实现任何具体协议（WebDAV/SFTP 等） |
| 定义 `Driver` 契约、能力位、注册与生命周期 | 不扫描目录、不解析标签、不做元数据补全 |

| 定义 `IDataSource` 跨层契约（实现在 03-06） | 不引用 `com.netease.*`、不引用播放内核 |

### 3.1 数据源层模型（对齐 OpenList-zig `model.zig`）

```kotlin
// ObjectType —— 对齐 model.zig:13-20 的编号，前端按 type 分支选图标，编号不可改
enum class ObjectType(val value: Int) {
    UNKNOWN(0), FOLDER(1), VIDEO(2), AUDIO(3), TEXT(4), IMAGE(5);
    companion object {
        // 集合加 _EXT 后缀，避免与同名枚举常量冲突
        // 判定顺序 AUDIO → VIDEO → IMAGE → TEXT（对齐 OpenList-zig utils.GetObjType）
        private val AUDIO_EXT = setOf("mp3","flac","ogg","m4a","wav","opus","wma","aac","ape","dsf","dff","aiff","wv","tak")
        private val VIDEO_EXT = setOf("mp4","mkv","avi","mov","rmvb","webm","flv")  // m3u8 移出：本项目按文本播放列表处理（03-03 §3.5，2026-11-04 修正）
        private val IMAGE_EXT = setOf("jpg","jpeg","png","gif","bmp","webp","avif","ico")
        private val TEXT_EXT  = setOf("lrc","strm","m3u","m3u8","pls","txt","nfo","srt","ass","cue")
        fun of(name: String, isDir: Boolean): ObjectType {
            if (isDir) return FOLDER
            val ext = extOf(name)
            if (ext.isEmpty()) return UNKNOWN
            return when {
                ext in AUDIO_EXT -> AUDIO
                ext in VIDEO_EXT -> VIDEO
                ext in IMAGE_EXT -> IMAGE
                ext in TEXT_EXT  -> TEXT
                else             -> UNKNOWN
            }
        }
        // 对齐 Go utils.Ext：取最后一个点之后的部分；".gitignore" → "gitignore"
        fun extOf(name: String): String = name.substringAfterLast('.', "").lowercase()
    }
}

data class HashInfo(val hashName: String, val hashValue: String)

data class Header(val name: String, val value: String)

data class Obj(
    val name: String,
    val size: Long = 0,
    val isDir: Boolean = false,
    val modified: Long = 0,          // Unix 秒；毫秒制驱动须 /1000
    val created: Long = 0,
    val sign: String = "",           // 不透明 token，link 时原样回传
    val thumb: String = "",
    val hashInfo: HashInfo? = null,
    val path: String = "",           // 驱动内相对根的路径（规范化后，见 §8）
    val priv: String? = null,        // 驱动私有载荷（如 itvsh 的真实服务端路径）
) {
    val type: ObjectType get() = ObjectType.of(name, isDir)
}

data class File(
    val name: String, val size: Long = 0, val isDir: Boolean = false,
    val modified: Long = 0, val created: Long = 0, val sign: String = "",
    val thumb: String = "", val hashInfo: HashInfo? = null,
) { companion object { fun from(o: Obj) = File(o.name, o.size, o.isDir, o.modified,
        o.created, o.sign, o.thumb, o.hashInfo) } }
```

### 3.2 `Link` —— ★全架构枢纽，三态必须严格互斥

```kotlin
/**
 * 交付形态（对齐 OpenList-zig model.zig:116）
 *  URL     : url 非空，内核/DMR 可直连
 *  LOCAL   : localPath 非空，本机文件，内核/SAF 直接读
 *  PROXIED : proxy=true，必须由 nanohttpd 本地代理转发（03-08）
 *  TEXT    : text 非空，内存虚拟文件内容（strm 虚拟目录）
 */
data class Link(
    val url: String = "",
    val headers: List<Header> = emptyList(),
    val localPath: String? = null,
    val localSize: Long = 0,
    val proxy: Boolean = false,
    val text: String? = null,
    val contentLength: Long = 0,
    val requireReference: Boolean = false,  // 需持有底层句柄（对齐 Go SyncClosers）
) {
    enum class Kind { URL, LOCAL, PROXIED, TEXT }

    val kind: Kind get() = when {
        !text.isNullOrEmpty()      -> Kind.TEXT
        localPath != null          -> Kind.LOCAL
        proxy || url.isEmpty()     -> Kind.PROXIED
        else                       -> Kind.URL
    }

    init {
        require(!(url.isNotEmpty() && localPath != null)) { "url 与 localPath 互斥" }
        require(!(!text.isNullOrEmpty() && !proxy)) { "text 必须伴随 proxy=true" }
        require(localPath == null || localSize >= 0) { "localSize 不得为负" }
        require(!url.isNotEmpty() || url.startsWith("http://") || url.startsWith("https://"))
            { "V0.1 直链只接受 http/https（PRD §5.2）" }
        if (requireReference) require(kind == Kind.PROXIED || kind == Kind.LOCAL)
            { "持有句柄的场景仅出现在 PROXIED/LOCAL" }
    }
}

data class LinkArgs(
    val method: String = "GET",
    val headers: List<Header> = emptyList(),
    val redirect: Boolean = false,   // true：先发一次请求取 302 Location 再返回（WebDAV）
)
```

**合法字段组合表**（03-02/03-03 实现时对照，03-06 分派时对照）：

| 场景 | kind | url | localPath | proxy | text | headers | requireReference |
|---|---|---|---|---|---|---|---|
| strm 直链 / 播放列表 URL 条目 | URL | http(s) | null | false | null | 可选签名参数 | false |
| WebDAV 302 后直链 | URL | http(s) | null | false | null | 可选 | false |
| 本地音乐 | LOCAL | "" | 绝对路径或 SAF Uri | false | null | 空 | false |
| WebDAV 鉴权流 / SFTP / FTP / NFS | PROXIED | "" 或原 URL | null | true | null | Basic/Digest | true |
| strm 虚拟目录文件内容 | TEXT | "" | null | true | URL 文本 | 空 | false |

> `Link.kind` 是 **唯一分派依据**：03-06 `DataSourceAdapter` 与 03-08 `CastUriResolver` 都只读它，不判断具体协议类型。

## 4. `Driver` 契约

```kotlin
interface Driver {
    val config: DriverConfig

    /** 校验配置 / 登录 / 恢复会话。失败抛 CloudMusicError；已初始化时由 DriverLifecycle 先 drop。 */
    suspend fun init(ctx: DriverCtx)

    /** 释放连接、关闭客户端。必须容忍「init 失败后被调用」，且可重复调用。 */
    fun drop()

    /** 列出 dir 的直接子项（不递归）。refresh=true 时绕过缓存。 */
    suspend fun list(dir: Obj, refresh: Boolean = false): List<Obj>

    /** 按驱动内相对路径取单文件；不存在返回 null（不是抛异常）。 */
    suspend fun get(path: String): Obj?

    /** ★统一交付点：把 Obj 变成可播放的 Link。 */
    suspend fun link(file: Obj, args: LinkArgs = LinkArgs()): Link
}

data class DriverConfig(
    val name: String,
    val onlyProxy: Boolean = false,     // 一切走代理
    val noLinkUrl: Boolean = false,     // 不返回直链
    val noCache: Boolean = false,       // 禁止缓存 list/link 结果
    val localSort: Boolean = true,      // 由驱动本地排序（不依赖服务端顺序）
    val preferProxy: Boolean = false,
) { fun mustProxy() = onlyProxy || noLinkUrl }

/** 请求级上下文（对齐 OpenList-zig driver.Ctx，值语义、不跨请求复用） */
data class DriverCtx(
    val tag: String = "",                       // 日志前缀，如 "[webdav:nas]"
    val caller: String = "",                    // "scan" | "play" | "test"
    val errDetail: MutableRef<String> = MutableRef(),       // 上游人类可读消息透传
    val coroutineContext: CoroutineContext = Dispatchers.IO,
) { fun setErrDetail(msg: String) { if (msg.isNotEmpty()) errDetail.value = msg } }

/** 极简请求级可变引用（对齐 OpenList-zig Ctx.err_detail 的值语义；不跨请求复用） */
class MutableRef<T>(@Volatile var value: T? = null)
```

**能力位语义**（驱动必须遵守，03-08 / 03-06 依赖）：

| 位 | 为 true 时 | 谁读 |
|---|---|---|
| `onlyProxy` | `link()` 一律返回 `proxy=true`，禁止直链 | 03-08 `CastUriResolver` |
| `noLinkUrl` | 不返回可直连 URL（等价 `mustProxy()`） | 03-06 `DataSourceAdapter` |
| `noCache` | `list/link` 结果不缓存（strm 直链会过期） | 03-04 `ScanCoordinator` |
| `localSort` | 驱动保证按 name 排序（默认 true） | 03-04 |
| `preferProxy` | 驱动级「投屏优先代理」默认（**默认 false**，决策 3 直链优先）；投屏生效值 = 用户 `forceWebdavProxy` **或** 驱动 `preferProxy`，为 true 时跳过 `directUrl()` 探测直接代理（03-08 §3.2，2026-11-04 修正，`10 §2 R6`） | 03-08 `CastUriResolver` |

### 4.1 注册表（Hilt Multibindings）

```kotlin
typealias DriverFactory = (Storage, Addition) -> Driver

object DriverRegistry {
    private val factories = ConcurrentHashMap<String, DriverFactory>()
    private val instances = ConcurrentHashMap<String, Driver>()

    fun register(type: String, factory: DriverFactory) { check(factories.putIfAbsent(type, factory) == null) }
    fun types(): Set<String> = factories.keys.toSet()
    fun create(type: String, storage: Storage, addition: Addition): Driver =
        factories[type]?.invoke(storage, addition) ?: throw CloudMusicError.DsUnknownDriver(type)
    fun get(storageId: String): Driver? = instances[storageId]
    fun put(storageId: String, d: Driver) { instances[storageId] = d }
    fun remove(storageId: String) { instances.remove(storageId)?.drop() }
}

/** ★全项目唯一的 Driver 注册点（放 :core-di；03-02 / 03-03 只提供构造，不另建 @Module） */
@Module @InstallIn(SingletonComponent::class)
object DataSourceModule {
    @Provides @IntoMap @StringKey("local")
    fun local(): DriverFactory = { s, a -> LocalDriver(s, a.raw) }
    @Provides @IntoMap @StringKey("webdav")
    fun webdav(): DriverFactory = { s, a -> WebDavDriver(s, a.raw) }
    @Provides @IntoMap @StringKey("strm")
    fun strm(): DriverFactory =
        { s, a -> StrmDriver(s, a.raw, DriverRegistry) }
    @Provides @IntoMap @StringKey("playlist")
    fun playlist(): DriverFactory =
        { s, a -> PlaylistDriver(s, a.raw, DriverRegistry) }
    // SFTP/FTP/NFS：V0.2+ 在此追加，不改其他模块
}
```

> 注册表只存 **factory**，不持有 `Addition` 的类型信息；配置由各驱动 `parse(Addition.raw)` 自行解释（对齐 OpenList `GetAddition()` 返回 `driver.Additional` 的做法）。
> `DriverFactory = (Storage, Addition) -> Driver`（与架构 §3.2 已回补的签名一致）：构造需要 `Storage.id` 用于委托型驱动定位底层数据源；`Addition.raw` 是 JSON 字符串，`LocalDriver` / `WebDavDriver` / `StrmDriver` / `PlaylistDriver` 的第二参数均为 `String`。
> **本 Module 是全项目唯一注册点**：03-02 §6、03-03 §2.1 均引用此处，不另建 `MountDriversModule`。

> **★ 2026-11-05 第十一批落地校准（`12 §20`）**：上方代码块是规格示意，与实际构造签名有两点差异，编解码时以实际代码为准：
> ① `WebDavDriver` 实际需要第三参数 `HttpStack`（生产用 `OkHttpStack(OkHttpClient())`，单测注入 fake）
> 及可选第四参数 `forceWebdavProxy`，见 `03-02 §4`；`:core-di/DataSourceModule` 已按实际签名提供。
> ② `local` 键对应 `:datasource-local`（SAF / MediaStore 双通道）尚未落地（`12 §11.4`），
> 故当前生产注册表只有 `webdav` / `strm` / `playlist` 三个键；落地时只需在 `DataSourceModule`
> 增加一个 `@IntoMap` 提供者，其余模块不改。契约测试 T16 的 4 键由纯 JVM fake 覆盖，不受影响。

### 4.2 生命周期状态机

```
NEW ──init 成功──▶ READY ──drop──▶ CLOSED
 │                    │                ▲
 └──init 失败──▶ FAILED ──drop─────────┘
                      │
                      └── 可重试 init（先 drop）
```

- `drop()` 必须幂等；`FAILED` 状态下调用不得抛异常。
- `init` 失败时把上游消息写入 `ctx.errDetail`，由 `Storage.errorMsg` 持久化，UI 显示「暂不可用」而非笼统「网络错误」（架构 §11）。
- 连接断开重连由**驱动内部**负责（参考 `sftp` 的 `clientReconnectOnConnectionError`），不暴露给上层。

## 5. `Link` 三态分派与 `IDataSource`

```kotlin
/** 跨层契约，实现在 :engine-bridge（03-06） */
interface IDataSource {
    /** 打开并返回总长度；-1 表示未知长度。offset 为起始偏移（Range）。 */
    fun open(offset: Long): Long
    /** 从 offset 读最多 len 字节到 buf，返回实际字节数；EOF 返回 0，-1 表示错误。 */
    fun readAt(offset: Long, buf: ByteArray, len: Int): Int
    fun close()
    fun mimeHint(): String
}
```

`Link.kind → IDataSource` 的分派（实现细节见 03-06 §7）：

| kind | 实现 | 取流方式 |
|---|---|---|
| URL | `HttpDataSource` | OkHttp，带 `Link.headers`，支持 Range |
| LOCAL | `FileDataSource` | SAF `Uri` / 文件描述符，`FileChannel.position(offset)` |
| PROXIED | `DriverDataSource` | 驱动提供带鉴权的流；本机播放时由本地代理回源消费（**经 nanohttpd 回环**，2026-11-06 修正，见 `16 §3` / 03-06 §5） |
| TEXT | `MemoryDataSource` | `text.encodeToByteArray()` 按偏移读 |

> 投屏（DLNA）另走一次决策：除 `LOCAL` 外全部直链，见 03-08 `CastUriResolver`。**不要**把本机播放的分派和投屏的分派混为一谈。
>
> **消费方勘误（2026-11-06，`13 §3` / `16 §3`）**：`IDataSource` 的消费方是**本地代理回源**（03-08 §8.3）与**扫描 Range 读**（§5.1 `ByteRangeCapability`），不再是引擎主取流路径——引擎舱经 `DataProvider` 吃 URL / 本地路径 / fd，native 自行拉流（03-06 §5）。

### 5.1 可选能力接口（03-02 / 03-04 定义，不改本契约）

| 接口 | 定义处 | 用途 | 未实现时 |
|---|---|---|---|
| `DirectLinkCapability.directUrl(file): String?` | 03-02 §4.6 | `CastUriResolver` 判断 `PROXIED` 曲目能否直推 DMR | 回落 nanohttpd 代理 |
| `ByteRangeCapability.readBytes(file, offset, length)` | 03-04 §3 | 扫描时只读文件头部做标签解析，避免整文件下载 | 阶段 B 跳过元数据补齐，保留文件名骨架 |

两者用 `as?` 探测，**不强制所有 `Driver` 实现**，也不改变 `Link` 三态与 `Driver` 的 5 个方法。新增协议时只需实现适合自己能力的可选接口。

```kotlin
// 共同基类 CloudMusicError 统一登记在 :core-common 的 ErrorCatalog（03-12）
```

**错误码映射规则**：驱动捕获底层异常后，若上游有人类可读消息，先 `ctx.setErrDetail(msg)` 再抛 `DsUpstream`；上层 `mapDriverError()` 优先取 `errDetail` 展示。禁止把 `IOException` 原样抛给 UI。

**工具函数约定**：`sha256()` 定义在 `:core-common` 的 `HashUtil`；`normalizePath()` 定义在 `:datasource-api` 的 `PathRule`（§8）。二者均为纯函数、无 Android 依赖。

## 7. ID 稳定性与去重

- `TrackId = sha256(namespace | 规范化相对路径).take(32)`；`namespace` 取 `Storage.id`（稳定 UUID），**不含 host / 端口 / 账号**。
- WebDAV 地址变更（换域名、换端口）→ 只改 `Addition`，`Storage.id` 不变 → `TrackId` 不变 → 收藏与播放记录保留（PRD §5.3）。
- 重复曲目**允许共存**（不同 `storageId` 或不同路径即不同 ID）；UI 的「合并重复项」只是展示层开关，不改数据。
- 挂载点重命名（`Storage.mountPath`）**不参与** ID 计算。

## 8. 路径与编码规范

| 规则 | 说明 |
|---|---|
| 规范化 | 统一 `/` 分隔、去除 `.` / `..`、折叠重复斜杠、结尾不加斜杠（目录除外）；`normalizePath()` 为纯函数，放 `:datasource-api` |
| 沙箱 | 任何外部传入路径先规范化，再**二次校验**是否仍位于根内（参考 `local.zig:7-13`）；越界抛 `DsNotSupport` |
| 前导点 | `.gitignore` 之类文件 `ext` 取 `gitignore`（对齐 Go `utils.Ext`） |
| 大小写 | 扩展名比较一律 `lowercase()`；文件名保留原样 |
| 中文编码 | FTP 驱动按 `Addition.encoding` 转换（参考 `drivers/ftp/util.go` mahonia），转换失败标 `DsEncoding` 并跳过该文件，不中断整轮扫描 |
| WebDAV href | 剥离命名空间前缀（`D:href`/`d:href`/`href` 等价）、尾斜杠归一化、URL-decode 后再拼路径 |
| 隐藏文件 | `Addition.showHidden=false` 时过滤以 `.` 开头的项；默认 true（对齐 `local.zig:34`） |
| 符号链接 | 默认不跟随（安全，参考 `local.zig:36-38`） |

## 10. 并发与取消

- `init / list / get / link` 全部 `suspend` 且**主线程安全**；内部自行切到 `Dispatchers.IO`，调用方不指定调度器。
- 所有网络操作必须响应协程取消（OkHttp `call.cancel()` 由 `suspendCancellableCoroutine` 绑定）；FTP/SFTP 连接用完 `Quit/Close`，取消时不留悬挂连接。
- `DriverRegistry` 用 `ConcurrentHashMap`；单个 `Driver` 实例**不保证线程安全**，多个协程同时 `list` 时由 03-04 `ScanCoordinator` 用 `Semaphore(4)` 限并发，驱动内部不额外加锁（除连接池自身）。
- `noCache=true` 的驱动（Strm）禁止在 `Driver` 内缓存 `list` 结果；其余可缓存但必须带 `Storage.modified` 版本号，配置一改即失效。
- `errDetail` 是**请求级**对象，禁止跨请求复用（对齐 OpenList-zig `driver.zig:47-73` 的注释：多线程写它属于数据竞争）。

## 14. 明确禁止事项

1. 禁止在 `:core-model` / `:datasource-api` 出现 `com.netease.*`、`OkHttp`、`Room`、`Context`。
2. 禁止让 `Driver` 返回「既是直链又是本地路径」的 `Link`（`init` 块已强制）。
3. 禁止在 `list()` 里做递归（递归由 03-04 `ScanCoordinator` 负责，驱动只返回直接子项）。
4. 禁止缓存带签名/时效的 strm 直链（`noCache=true`）。
5. 禁止因一次播放失败删除 `Track`（只改 `availability`）。
6. 禁止照抄竞品 Manifest 的权限或把凭据写进日志。
